CREATE DATABASE eval_back;

USE eval_back;

CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    nom VARCHAR(50),
    prenom VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    role ENUM('user','admin') DEFAULT 'user'
);



CREATE TABLE concert (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ville VARCHAR(50) NOT NULL,
    lieu VARCHAR(50) NOT NULL,
    date DATE NOT NULL,
    heure TIME NOT NULL DEFAULT '20:00:00'
);



CREATE TABLE places (

    id INT AUTO_INCREMENT PRIMARY KEY,
    id_concert INT UNIQUE,
    stock INT CHECK(stock >=0),
    prix DECIMAL(10,2),
    CONSTRAINT places_ibfk_1
        FOREIGN KEY (id_concert)
        REFERENCES concert(id)
        ON DELETE CASCADE
);



CREATE TABLE commandes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_user INT NOT NULL, 
    date_commande TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    total DECIMAL(10,2),
    CONSTRAINT commandes_ibfk_1
        FOREIGN KEY (id_user)
        REFERENCES users(id)
);



CREATE TABLE billets (
    id INT AUTO_INCREMENT PRIMARY KEY,
    id_concert INT NOT NULL ,
    id_commande INT NOT NULL,
    quantity INT NOT NULL,
    prix DECIMAL(10,2),

    CONSTRAINT billets_ibfk_1
        FOREIGN KEY (id_concert)
        REFERENCES concert(id),

    CONSTRAINT billets_ibfk_2
        FOREIGN KEY (id_commande)
        REFERENCES commandes(id)
);



CREATE TABLE cart_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    concert_id INT NOT NULL,
    quantity INT NOT NULL CHECK(quantity > 0),
    price DECIMAL(10,2) NOT NULL,
    total DECIMAL(10,2) NOT NULL,
    status ENUM(
        'active',
        'reserved',
        'converted'
    ) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    CONSTRAINT cart_items_ibfk_1
        FOREIGN KEY (user_id)
        REFERENCES users(id),

    CONSTRAINT cart_items_ibfk_2
        FOREIGN KEY (concert_id)
        REFERENCES concert(id)

);
